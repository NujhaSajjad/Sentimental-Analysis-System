import os
import sys

# Sirf in extensions ko process karenge (text/code files jo .txt nahi hain)
ALLOWED_EXTENSIONS = {
    '.py', '.js', '.jsx', '.ts', '.tsx', '.html', '.htm', '.css', '.scss', '.sass',
    '.json', '.yaml', '.yml', '.xml', '.csv', '.sql', '.md', '.markdown',
    '.cpp', '.c', '.h', '.hpp', '.java', '.kt', '.go', '.rs', '.php', '.rb',
    '.sh', '.bat', '.ps1', '.ini', '.cfg', '.toml', '.env', '.gitignore',
    '.gitattributes', '.log', '.vue', '.svelte'
    # aur jitni chahiye yahan add kar sakte ho
}

# In extensions ko hamesha skip (images, videos, aur .txt)
SKIP_EXTENSIONS = {
    '.txt',                 # existing .txt files ignore
    '.jpg', '.jpeg', '.png', '.gif', '.webp', '.bmp', '.tiff', '.ico', '.svg',  # images
    '.mp4', '.mkv', '.avi', '.mov', '.webm', '.flv', '.wmv', '.mpeg',         # videos
    '.pdf', '.doc', '.docx', '.xls', '.xlsx', '.ppt', '.pptx',                # documents
    '.zip', '.rar', '.7z', '.tar', '.gz',                                     # archives
    '.exe', '.dll', '.bin', '.dat', '.db'                                     # binaries + databases
}

def extract_code_text_files(root_dir=None):
    if root_dir is None:
        root_dir = os.getcwd()
    
    output_folder = os.path.join(root_dir, "extracted_code_text")
    os.makedirs(output_folder, exist_ok=True)
    
    print(f"Scanning folder: {root_dir}")
    print(f"Output folder:   {output_folder}")
    print("→ .txt files, images (.jpg, .png...), videos (.mp4...), binaries skip kiye jayenge")
    print("-" * 80)
    
    processed = 0
    skipped = 0
    
    for dirpath, dirnames, filenames in os.walk(root_dir):
        # output folder ko walk se hata do taake andar na jaye
        dirnames[:] = [d for d in dirnames if d != "extracted_code_text"]
        
        for filename in filenames:
            file_path = os.path.join(dirpath, filename)
            rel_path = os.path.relpath(file_path, root_dir)
            
            name_lower = filename.lower()
            _, ext = os.path.splitext(name_lower)
            
            # Explicit skip rules
            if ext in SKIP_EXTENSIONS:
                print(f"skip → {rel_path}")
                skipped += 1
                continue
            
            # Sirf allowed list mein ho to process
            if ext not in ALLOWED_EXTENSIONS:
                print(f"skip (not in allowed list) → {rel_path}")
                skipped += 1
                continue
            
            # Output .txt file banayenge
            txt_filename = filename + ".txt"
            txt_path = os.path.join(output_folder, txt_filename)
            
            # Duplicate name handle
            counter = 1
            base, _ = os.path.splitext(txt_filename)
            while os.path.exists(txt_path):
                txt_path = os.path.join(output_folder, f"{base}_{counter}.txt")
                counter += 1
            
            try:
                with open(file_path, "r", encoding="utf-8") as f:
                    content = f.read()
                
                with open(txt_path, "w", encoding="utf-8") as out:
                    out.write(f"Relative path: {rel_path}\n")
                    out.write("=" * 80 + "\n\n")
                    out.write(content)
                    if not content.endswith("\n"):
                        out.write("\n")
                
                print(f"✓ {rel_path:<50} → {os.path.basename(txt_path)}")
                processed += 1
                
            except UnicodeDecodeError:
                print(f"skip (not readable text) → {rel_path}")
                skipped += 1
            except Exception as e:
                print(f"error → {rel_path}  ({str(e)})")
                skipped += 1
    
    print("\n" + "="*80)
    print(f"Done!   Processed: {processed} files")
    print(f"Skipped: {skipped} files (images / videos / .txt / others)")
    print(f"Check folder → {output_folder}")


if __name__ == "__main__":
    if len(sys.argv) > 1:
        extract_code_text_files(sys.argv[1])
    else:
        extract_code_text_files()
