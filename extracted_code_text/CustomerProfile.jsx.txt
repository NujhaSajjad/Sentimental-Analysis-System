import React, { useState, useEffect } from 'react';
import { 
  User, Phone, Mail, MapPin, Building, Calendar, 
  TrendingUp, AlertTriangle, CheckCircle, XCircle 
} from 'lucide-react';
import axios from 'axios';

const API_URL = process.env.REACT_APP_API_URL || 'http://localhost:3000';

const CustomerProfile = ({ customer, onViewCalls, onClose }) => {
  const [profile, setProfile] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (customer && customer.cnic) {
      fetchCustomerProfile(customer.cnic);
    }
  }, [customer]);

  const fetchCustomerProfile = async (cnic) => {
    try {
      setLoading(true);
      const response = await axios.get(`${API_URL}/api/customers/${cnic}/profile`);
      
      if (response.data.success) {
        setProfile(response.data.profile);
      }
    } catch (error) {
      console.error('Failed to fetch profile:', error);
    } finally {
      setLoading(false);
    }
  };

  const getChurnRiskClass = (risk) => {
    if (!risk) return 'low';
    const lowerRisk = risk.toLowerCase();
    if (lowerRisk === 'critical' || lowerRisk === 'high') return 'high';
    if (lowerRisk === 'medium') return 'medium';
    return 'low';
  };

  const getSentimentColor = (sentiment) => {
    if (!sentiment) return '#6B7280';
    const lowerSentiment = sentiment.toLowerCase();
    if (lowerSentiment === 'positive') return '#10B981';
    if (lowerSentiment === 'negative') return '#EF4444';
    return '#F59E0B';
  };

  const formatDate = (date) => {
    if (!date) return 'N/A';
    return new Date(date).toLocaleDateString('en-US', {
      year: 'numeric',
      month: 'short',
      day: 'numeric'
    });
  };

  if (loading) {
    return (
      <div className="customer-profile-modal">
        <div className="profile-loading">
          <div className="spinner" />
          <p>Loading customer profile...</p>
        </div>
      </div>
    );
  }

  if (!profile) {
    return null;
  }

  return (
    <div className="customer-profile-modal" onClick={onClose}>
      <div className="profile-content" onClick={(e) => e.stopPropagation()}>
        {/* Header */}
        <div className="profile-header">
          <div className="profile-avatar">
            <User size={32} />
          </div>
          <div className="profile-title">
            <h2>{profile.full_name || 'Unknown Customer'}</h2>
            <p className="profile-subtitle">Complete Customer Intelligence</p>
          </div>
          <button className="close-button" onClick={onClose}>Ã—</button>
        </div>

        {/* Quick Stats */}
        <div className="profile-quick-stats">
          <div className="quick-stat">
            <div className="stat-icon" style={{ background: '#EEF2FF' }}>
              <Phone size={20} style={{ color: '#6366F1' }} />
            </div>
            <div>
              <div className="stat-value">{profile.total_calls || 0}</div>
              <div className="stat-label">Total Calls</div>
            </div>
          </div>
          <div className="quick-stat">
            <div className="stat-icon" style={{ background: getSentimentColor(profile.overall_sentiment) + '20' }}>
              <TrendingUp size={20} style={{ color: getSentimentColor(profile.overall_sentiment) }} />
            </div>
            <div>
              <div className="stat-value">{profile.overall_sentiment || 'N/A'}</div>
              <div className="stat-label">Overall Sentiment</div>
            </div>
          </div>
          <div className="quick-stat">
            <div className={`stat-icon churn-${getChurnRiskClass(profile.churn_risk)}`}>
              <AlertTriangle size={20} />
            </div>
            <div>
              <div className="stat-value">{profile.churn_risk || 'Low'}</div>
              <div className="stat-label">Churn Risk</div>
            </div>
          </div>
        </div>

        {/* Contact Information */}
        <div className="profile-section">
          <h3 className="section-title">Contact Information</h3>
          <div className="info-grid">
            <div className="info-item">
              <div className="info-icon">
                <Phone size={16} />
              </div>
              <div>
                <div className="info-label">Phone Number</div>
                <div className="info-value">{profile.phone_number}</div>
              </div>
            </div>
            <div className="info-item">
              <div className="info-icon">
                <Mail size={16} />
              </div>
              <div>
                <div className="info-label">Email</div>
                <div className="info-value">{profile.email || 'Not provided'}</div>
              </div>
            </div>
            <div className="info-item">
              <div className="info-icon">
                <User size={16} />
              </div>
              <div>
                <div className="info-label">CNIC</div>
                <div className="info-value">{profile.cnic}</div>
              </div>
            </div>
            <div className="info-item">
              <div className="info-icon">
                <MapPin size={16} />
              </div>
              <div>
                <div className="info-label">City</div>
                <div className="info-value">{profile.city || 'Not provided'}</div>
              </div>
            </div>
            {profile.company_name && (
              <div className="info-item">
                <div className="info-icon">
                  <Building size={16} />
                </div>
                <div>
                  <div className="info-label">Company</div>
                  <div className="info-value">{profile.company_name}</div>
                </div>
              </div>
            )}
            <div className="info-item">
              <div className="info-icon">
                <Calendar size={16} />
              </div>
              <div>
                <div className="info-label">Customer Since</div>
                <div className="info-value">{formatDate(profile.created_at)}</div>
              </div>
            </div>
          </div>
        </div>

        {/* Call Statistics */}
        <div className="profile-section">
          <h3 className="section-title">Call Analytics</h3>
          <div className="analytics-grid">
            <div className="analytics-card">
              <div className="analytics-label">First Call</div>
              <div className="analytics-value">{formatDate(profile.first_call_date)}</div>
            </div>
            <div className="analytics-card">
              <div className="analytics-label">Last Call</div>
              <div className="analytics-value">{formatDate(profile.last_call_date)}</div>
            </div>
            <div className="analytics-card">
              <div className="analytics-label">Avg Quality Score</div>
              <div className="analytics-value">
                {profile.avg_quality_score ? `${Math.round(parseFloat(profile.avg_quality_score))}/100` : 'N/A'}
              </div>
            </div>
            <div className="analytics-card">
              <div className="analytics-label">Avg CSAT</div>
              <div className="analytics-value">
                {profile.avg_csat ? `${parseFloat(profile.avg_csat).toFixed(1)}/5.0` : 'N/A'}
              </div>
            </div>
          </div>
        </div>

        {/* Sentiment Analysis */}
        {profile.overall_sentiment_score && (
          <div className="profile-section">
            <h3 className="section-title">Sentiment Analysis</h3>
            <div className="sentiment-bar-container">
              <div className="sentiment-labels">
                <span>Negative</span>
                <span>Neutral</span>
                <span>Positive</span>
              </div>
              <div className="sentiment-bar">
                <div 
                  className="sentiment-fill"
                  style={{ 
                    width: `${((parseFloat(profile.overall_sentiment_score) + 100) / 200) * 100}%`,
                    background: getSentimentColor(profile.overall_sentiment)
                  }}
                />
              </div>
              <div className="sentiment-score">
                Score: {parseFloat(profile.overall_sentiment_score).toFixed(0)}
              </div>
            </div>
          </div>
        )}

        {/* Resolution Status */}
        {profile.total_calls > 0 && (
          <div className="profile-section">
            <h3 className="section-title">Issue Resolution</h3>
            <div className="resolution-stats">
              <div className="resolution-item">
                <CheckCircle size={16} style={{ color: '#10B981' }} />
                <span>Resolved Issues</span>
                <strong>{profile.resolved_issues || 0}</strong>
              </div>
              <div className="resolution-item">
                <XCircle size={16} style={{ color: '#EF4444' }} />
                <span>Pending Issues</span>
                <strong>{profile.pending_issues || 0}</strong>
              </div>
            </div>
          </div>
        )}

        {/* Actions */}
        <div className="profile-actions">
          <button 
            className="btn btn-primary"
            onClick={() => onViewCalls(profile.customer_id)}
          >
            View Call History ({profile.total_calls})
          </button>
          <button className="btn btn-secondary" onClick={onClose}>
            Close
          </button>
        </div>
      </div>
    </div>
  );
};

export default CustomerProfile;
