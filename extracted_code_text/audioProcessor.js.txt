// audioProcessor.js - Audio Processing Utilities for Long Calls
const fs = require('fs');
const path = require('path');
const ffmpeg = require('fluent-ffmpeg');

/**
 * Split long audio into manageable chunks
 * @param {string} audioPath - Path to the audio file
 * @param {number} chunkDurationSeconds - Duration of each chunk (default: 300s = 5min)
 * @returns {Promise<Array>} Array of chunk objects with path and metadata
 */
async function splitAudioIntoChunks(audioPath, chunkDurationSeconds = 300) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    const outputDir = path.join(path.dirname(audioPath), 'chunks');
    
    // Create chunks directory
    if (!fs.existsSync(outputDir)) {
      fs.mkdirSync(outputDir, { recursive: true });
    }
    
    console.log(`✂️ Splitting audio: ${path.basename(audioPath)}`);
    
    // Get audio duration first
    ffmpeg.ffprobe(audioPath, (err, metadata) => {
      if (err) {
        console.error('❌ FFprobe error:', err);
        return reject(new Error(`Failed to analyze audio file: ${err.message}`));
      }
      
      const duration = metadata.format.duration;
      const numChunks = Math.ceil(duration / chunkDurationSeconds);
      
      console.log(`📊 Audio duration: ${Math.floor(duration / 60)}m ${Math.floor(duration % 60)}s`);
      console.log(`📦 Creating ${numChunks} chunks of ~${chunkDurationSeconds}s each`);
      
      if (numChunks === 1) {
        // Audio is already short enough
        console.log('✅ Audio is short enough, no chunking needed');
        resolve([{
          index: 0,
          path: audioPath,
          startTime: 0,
          duration: duration
        }]);
        return;
      }
      
      let chunksProcessed = 0;
      
      // Create each chunk
      for (let i = 0; i < numChunks; i++) {
        const startTime = i * chunkDurationSeconds;
        const chunkPath = path.join(outputDir, `chunk_${i}.mp3`);
        
        console.log(`🔄 Processing chunk ${i + 1}/${numChunks}...`);
        
        ffmpeg(audioPath)
          .setStartTime(startTime)
          .setDuration(chunkDurationSeconds)
          .audioCodec('libmp3lame')
          .audioBitrate('128k')
          .output(chunkPath)
          .on('end', () => {
            chunks.push({
              index: i,
              path: chunkPath,
              startTime: startTime,
              duration: Math.min(chunkDurationSeconds, duration - startTime)
            });
            
            chunksProcessed++;
            console.log(`✅ Chunk ${i + 1}/${numChunks} created`);
            
            if (chunksProcessed === numChunks) {
              console.log('✅ All chunks created successfully');
              resolve(chunks.sort((a, b) => a.index - b.index));
            }
          })
          .on('error', (err) => {
            console.error(`❌ Error creating chunk ${i}:`, err);
            reject(new Error(`Failed to create chunk ${i}: ${err.message}`));
          })
          .run();
      }
    });
  });
}

/**
 * Check if audio file is too long (> threshold)
 * @param {string} audioPath - Path to the audio file
 * @param {number} thresholdSeconds - Threshold in seconds (default: 300s = 5min)
 * @returns {Promise<boolean>} True if audio is longer than threshold
 */
async function isLongAudio(audioPath, thresholdSeconds = 300) {
  return new Promise((resolve, reject) => {
    ffmpeg.ffprobe(audioPath, (err, metadata) => {
      if (err) {
        console.error('❌ FFprobe error:', err);
        return reject(new Error(`Failed to analyze audio: ${err.message}`));
      }
      
      const duration = metadata.format.duration;
      const isLong = duration > thresholdSeconds;
      
      console.log(`⏱️ Audio duration: ${Math.floor(duration / 60)}m ${Math.floor(duration % 60)}s`);
      console.log(`📏 Threshold: ${Math.floor(thresholdSeconds / 60)}m`);
      console.log(`🎯 Is long: ${isLong ? 'YES - will chunk' : 'NO - direct processing'}`);
      
      resolve(isLong);
    });
  });
}

/**
 * Get audio file metadata
 * @param {string} audioPath - Path to the audio file
 * @returns {Promise<Object>} Audio metadata
 */
async function getAudioMetadata(audioPath) {
  return new Promise((resolve, reject) => {
    ffmpeg.ffprobe(audioPath, (err, metadata) => {
      if (err) {
        return reject(err);
      }
      
      const audioStream = metadata.streams.find(s => s.codec_type === 'audio');
      
      resolve({
        duration: metadata.format.duration,
        size: metadata.format.size,
        bitRate: metadata.format.bit_rate,
        format: metadata.format.format_name,
        codec: audioStream ? audioStream.codec_name : 'unknown',
        sampleRate: audioStream ? audioStream.sample_rate : null,
        channels: audioStream ? audioStream.channels : null
      });
    });
  });
}

/**
 * Cleanup temporary chunk files and directory
 * @param {string} chunkDir - Directory containing chunk files
 */
function cleanupChunks(chunkDir) {
  try {
    if (fs.existsSync(chunkDir)) {
      const files = fs.readdirSync(chunkDir);
      
      // Delete all files in directory
      files.forEach(file => {
        const filePath = path.join(chunkDir, file);
        fs.unlinkSync(filePath);
      });
      
      // Remove directory
      fs.rmdirSync(chunkDir);
      console.log('🧹 Cleaned up audio chunks');
    }
  } catch (error) {
    console.error('⚠️ Error cleaning up chunks:', error.message);
    // Don't throw - cleanup is optional
  }
}

/**
 * Convert audio to standard format for processing
 * @param {string} audioPath - Path to the audio file
 * @param {string} outputPath - Path for converted file
 * @returns {Promise<string>} Path to converted file
 */
async function convertAudioFormat(audioPath, outputPath = null) {
  return new Promise((resolve, reject) => {
    if (!outputPath) {
      const dir = path.dirname(audioPath);
      const name = path.basename(audioPath, path.extname(audioPath));
      outputPath = path.join(dir, `${name}_converted.mp3`);
    }
    
    console.log(`🔄 Converting audio to standard format...`);
    
    ffmpeg(audioPath)
      .audioCodec('libmp3lame')
      .audioBitrate('128k')
      .audioChannels(1) // Mono for better Whisper performance
      .audioFrequency(16000) // 16kHz sample rate
      .output(outputPath)
      .on('end', () => {
        console.log('✅ Audio converted successfully');
        resolve(outputPath);
      })
      .on('error', (err) => {
        console.error('❌ Conversion error:', err);
        reject(new Error(`Failed to convert audio: ${err.message}`));
      })
      .run();
  });
}

/**
 * Estimate processing time based on audio duration
 * @param {number} durationSeconds - Audio duration in seconds
 * @returns {Object} Estimated processing times
 */
function estimateProcessingTime(durationSeconds) {
  // Whisper typically processes at ~0.5x real-time for base model
  const whisperTimeSeconds = durationSeconds * 0.5;
  
  // AI analysis ~2-5 seconds per call
  const aiTimeSeconds = 5;
  
  // Total estimate
  const totalSeconds = whisperTimeSeconds + aiTimeSeconds;
  
  return {
    whisperMinutes: Math.ceil(whisperTimeSeconds / 60),
    aiSeconds: aiTimeSeconds,
    totalMinutes: Math.ceil(totalSeconds / 60),
    estimatedCompletion: new Date(Date.now() + totalSeconds * 1000)
  };
}
